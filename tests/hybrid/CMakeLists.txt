cmake_minimum_required(VERSION 3.10)

add_executable(hybrid_ex1_balance.out hybrid_ex1_balance.cpp)
target_link_libraries(hybrid_ex1_balance.out ${PROJECT_NAME})

set(balance_options -snes_atol 1e-12 -snes_rtol 1e-12 -snes_stol 0)
add_test(NAME hybrid_ex1_balance COMMAND ${MPI_DIR}/bin/mpiexec --bind-to none -np 1 ./hybrid_ex1_balance.out ${balance_options})
set_property(TEST hybrid_ex1_balance PROPERTY LABELS sanity impls hybrid)
set_property(TEST hybrid_ex1_balance PROPERTY RESOURCE_LOCK hybrid)
foreach(axis x y z)
  set(test hybrid_ex1_balance_mpi_${axis}2)
  add_test(NAME ${test} COMMAND ${MPI_DIR}/bin/mpiexec --bind-to none -np 2 ./hybrid_ex1_balance.out ${balance_options} -da_processors_${axis} 2)
  set_property(TEST ${test} PROPERTY LABELS sanity impls hybrid mpi)
  set_property(TEST ${test} PROPERTY RESOURCE_LOCK hybrid)
endforeach()

add_executable(hybrid_ex2_fms.out hybrid_ex2_fms.cpp)
target_link_libraries(hybrid_ex2_fms.out ${PROJECT_NAME})
add_test(NAME hybrid_ex2_fms COMMAND ${MPI_DIR}/bin/mpiexec --bind-to none -np 1 ./hybrid_ex2_fms.out)
set_property(TEST hybrid_ex2_fms PROPERTY LABELS regression impls hybrid)
set_property(TEST hybrid_ex2_fms PROPERTY RESOURCE_LOCK "hybrid")

add_test(NAME hybrid_ex2_fms_plot COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/tools/hybrid_ex2_fms.py)
set_property(TEST hybrid_ex2_fms_plot PROPERTY DEPENDS hybrid_ex2_fms)
set_property(TEST hybrid_ex2_fms_plot PROPERTY LABELS regression impls hybrid)
set_property(TEST hybrid_ex2_fms_plot PROPERTY ENVIRONMENT "MPLBACKEND=Agg")
